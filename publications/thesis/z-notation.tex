\nomenclature[c-23-pol]{$\pol(s)$}{action taken in state $s$ by a deterministic policy $\pol$}

\nomenclature[c-24-order]{$\pol \preceq \pol'$}{policy $\pol'$ is at least as good as policy $\pol$}
