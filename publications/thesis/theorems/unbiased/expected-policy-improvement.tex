\begin{lemma}
\label{thm: expected policy improvement}
Given an initial condition $q$, let $(\pol_k)$ be the sequence of policies generated by system \eqref{eq: update unbiased over policies}.
Under \autoref{asp: unbiased over policies},
% consecutive policies satisfy in all states
% if policy $\pol_\kpo$ differs from policy $\pol_\kpz$, 
for every state where $\pol_\kpz(s) \neq \pol_\kpo(s)$
\begin{align} 
    q_{\pol_\kpz}(s, \pol_\kpz(s)) + w_\kpz(s, \pol_\kpz(s))
    \ < \
    q_{\pol_\kpz}(s, \pol_\kpo(s)) + w_\kpz(s, \pol_\kpo(s)).
\end{align}
% where $d_\kpz$ is noise with zero bias and bounded variance.
\end{lemma}

\begin{proof}

%     + \lrate_\kpz \update_{\pol_\kpz}(s, \pol_\kpo(s)) \big( q_{\pol_\kpz}(s, \pol_\kpo(s)) + w_\kpz(s, \pol_\kpo(s)) \big)
%     % q_\kpo(\pol_\kpo)
%     % &\ = \
%     % (1 - \lrate_\kpz \update_{\pol_\kpz}(\pol_\kpo)) q_\kpz(\pol_\kpo)
%     % + \lrate_\kpz \update_{\pol_\kpz}(\pol_\kpo) \big( q_{\pol_\kpz}(\pol_\kpo) + w_\kpz(\pol_\kpo) \big)
% \end{align*}

Since the policies $\pol_\kpz$ and $\pol_\kpo$ are greedy on $q_\kpz$ and $q_\kpo$, respectively, 
and \autoref{asp: unbiased over policies} asserts that 
$\update_{\pol_\kpz}(s, \pol_\kpz(s)) = \update_{\pol_\kpz}(s, \pol_\kpo(s))$, 
% $\update_{\pol_\kpz} \rvert_{\pol_\kpz} = \update_{\pol_\kpz} \rvert_{\pol_\kpo}$,
for every state where $\pol_\kpz(s) \neq \pol_\kpo(s)$ it holds that
\begin{align*}
    (1 - \lrate_\kpz \update_{\pol_\kpz}(s, \pol_\kpz(s))) q_\kpz(s, \pol_\kpz(s)) &> (1 - \lrate_\kpz \update_{\pol_\kpz}(s, \pol_\kpo(s))) q_\kpz(s, \pol_\kpo(s)) 
    \\
    q_\kpo(s, \pol_\kpz(s)) &< q_\kpo(s, \pol_\kpo(s))
    \\
\end{align*}
Therefore,
\begin{align*}
    \lrate_\kpz \update_{\pol_\kpz}(s, \pol_\kpz(s)) &\big( q_{\pol_\kpz}(s, \pol_\kpz(s)) + w_\kpz(s, \pol_\kpz(s)) \big)
    \\&= q_\kpo(s, \pol_\kpz(s)) - (1 - \lrate_\kpz \update_{\pol_\kpz}(s, \pol_\kpz(s))) q_\kpz(s, \pol_\kpz(s))
    \\
    &< q_\kpo(s, \pol_\kpo(s)) - (1 - \lrate_\kpz \update_{\pol_\kpz}(s, \pol_\kpz(s))) q_\kpz(s, \pol_\kpo(s))
    \\
    &= \lrate_\kpz \update_{\pol_\kpz}(s, \pol_\kpz(s)) \big( q_{\pol_\kpz}(s, \pol_\kpo(s)) + w_\kpz(s, \pol_\kpo(s)) \big)
\end{align*}

\begin{align}
    q_{\pol_\kpz}(s, \pol_\kpz(s)) + w_\kpz(s, \pol_\kpz(s))
