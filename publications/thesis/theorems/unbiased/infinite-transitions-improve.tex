\begin{lemma}
\label{thm: policy improvement if visit infinitely often}
Let the sequence $(\pol_\kpz)$ be generated by system \eqref{eq: update unbiased over policies} under \autoref{asp: unbiased over policies}.
If there exists a pair of policies $\pol, \pol_+ \in \polset$ such that
\begin{align}
    \forall \ M \geq 0, \
    \exists \ k > M \
    : \
    \pol_\kpz = \pol 
    \text{ and } 
    \pol_\kpo = \pol_+ ,
\end{align}
% if system \eqref{eq: update unbiased over policies} transitions infinitely often from a policy $\pol$ to another policy $\pol_+$, 
then policy $\pol_+$ is strictly better than policy $\pol$ in at least one state.
\end{lemma}
% 
\begin{proof}
\autoref{thm: all inequalities when switching policy}
shows that system \eqref{eq: update unbiased over policies} encounters an infinite number of estimates $\qval_\kpz \in \region(\pol)$ that satisfy the inequality
\begin{align}
\label{eq: inequality at transition on pol}
    \update_\kpz(s, \pol) ( \qval_\pol(s, \pol) - \qval_{\kpz}(s, \pol) )
    &< 
    \update_\kpz(s, \pol_+) ( \qval_\pol(s, \pol_+) - \qval_{\kpz}(s, \pol) )
\end{align}
for all states $s$ where $\pol(s) \neq \pol_+(s)$.
Since, by \autoref{asp: unbiased over policies}, the biases towards policies $\pol$ and $\pol_+$ are equal, namely $\update_\kpz(s, \pol) = \update_\kpz(s, \pol_+)$, inequality \eqref{eq: inequality at transition on pol} simplifies to
\begin{align}
\label{eq: pit in unbiaed lemma for improvement}
    \qval_\pol(s, \pol) < \qval_\pol(s, \pol_+)
\end{align}
for all states $s$ where $\pol(s) \neq \pol_+(s)$.
Equation \eqref{eq: pit in unbiaed lemma for improvement} satisfies the \hyperref[thm: policy improvement theorem]{policy improvement theorem}, which implies that policy $\pol_+$ is strictly better than policy $\pol$ in at least one state.
