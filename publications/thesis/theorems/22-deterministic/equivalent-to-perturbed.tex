\begin{lemma}
\label{thm: mc-opi behaves like perturbed sys}
    If the learning rates $(\lrate_k)_{k \geq 0}$ and update distributions $(\update_k)_{k \geq 0}$ satisfy the modified Robbins-Monro conditions (\autoref{asp: stochastic conditions on effective learning rate})
    and 
    the action values $q_{\best{\pol}}$ are globally asymptotically stable for 
    % the difference inclusion \eqref{eq: mcopi determ},
    the difference inclusion \eqref{eq: mcopi determ time inv} with respect to $\omega(q, k) = \|\qopt - q\|$,
    then for every $\epsilon>0$, $\eta>0$
    satisfying $\eta < \epsilon$,
    and sequence $(\lrate_k \update_k w_k)_{k \geq 0}$ generated by MC-O-PI, as defined in \eqref{eq: mcopi stochastic},
    there exists almost surely a time step $K$ such that for all $k \geq K$ and $q \in \qset(\epsilon, \eta)$ we have
    \begin{align}
        \lrate_k \update_k w_k \in d(q) \ball 
    \end{align}
    where $d : \qset \to \real_{\geq 0}$ is the perturbation radius defined in \autoref{thm: perturbed di converges}.

    % \begin{align}
    %     \prob \big( \exists K, \forall k \geq K, \forall q \in \qset(\epsilon, \eta) : \lrate_k \update_k w_k \in d(q) \ball \big) = 1 .
    % \end{align}

    % gets arbitrarily close infinitely often
    % for all 
    % \begin{align}
    %     \prob \big( \forall K, \exists k \geq K : \|\qopt - q_{k} \| \leq \epsilon \big) = 1 .
    % \end{align}
    
    % a solution of MC-O-PI in the region
    % \begin{align*}
    %     \{ q \in \qset : \| q_{\best{\pol}} - q \| \geq \epsilon \} ,
    % \end{align*}
    % there exists a time step $K \geq 0$ such that for all $k \geq K$
    % is also a solution of the perturbed difference inclusion \eqref{eq: perturbed inclusion}.

    % and every compact domain $\qset'$ that does not contain the set
    % \begin{align*}
    %     \{ q \in \qset : \| q_{\best{\pol}} - q \| < \epsilon \} ,
    % \end{align*}
    % there exists a time step $K$ such for all $k>K$, MC-OPI is equivalent to the perturbed difference inclusion \eqref{eq: perturbed inclusion} over $\qset'$ \textcolor{red}{w.p.o.}.

\begin{proof}
% Since the difference inclusion \eqref{eq: mcopi determ time inv} is globally stable,
The continuous function $d$ of \autoref{thm: perturbed di converges} is strictly positive for all $q \neq \qopt$.
% such that the perturbed difference inclusion \eqref{eq: perturbed inclusion} is globally stable. 
Further, for every $\epsilon>0$ and $\eta>0$ satisfying $\eta < \epsilon$, the set $\qset(\epsilon, \eta)$ is non-empty, compact and excludes $\qopt$.
Thus the minimum of $d$ over $\qset(\epsilon, \eta)$
% \begin{align}
%     \min_{q \in \qset'} d(q)
% \end{align}
is well-defined and strictly positive.

Further, \autoref{thm: effects of noise decrease} indicates that, under the modified Robbins-Monro conditions, the sequence $(\lrate_k \update_k w_k)_{k \geq 0}$ generated by MC-O-PI converges to zero with probability one for every state-action pair. 
Thus, for every $\epsilon>0$ and $\eta>0$ satisfying $\eta < \epsilon$, there exists almost surely a time step $K$ such that, for all $k \geq K$, we have
\begin{align*}
    % \prob \big( \exists K, \forall k \geq K, \forall q \in \qset(\epsilon, \eta) : 
    \max_{s,a} | \lrate_k \update_k(s,a) w_k(s,a) | < \min_{\qval \in \qset(\epsilon, \eta)} d(q) .
    % \big) = 1 ,
\end{align*}
and consequently $\lrate_k \update_k w_k \in d(q) \ball$ for all $q \in \qset(\epsilon, \eta)$.
% \begin{align*}
%     \lrate_k \update_k w_k \in d(q) \ball .
% \end{align*}
\end{proof}
