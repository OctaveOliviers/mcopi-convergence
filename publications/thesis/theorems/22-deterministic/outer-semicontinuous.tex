\begin{definition}[Outer semicontinuity]
    The set-valued map $F: \xset \rightrightarrows \xset$ is outer semicontinuous 
    % on $\xset$ 
    if and only if,
    for every $x \in \xset$ and every sequence $(x_k)_{k \geq 0}$ that converges to $x$,
    every sequence $(f_k \in F(x_k))_{k \geq 0}$ satisfies
    \begin{align*}
        \lim_{k \to \infty} f_k \in F(x) .
    \end{align*}
    
    
\end{definition}
